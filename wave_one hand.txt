from pymycobot.mybuddy import MyBuddy
import time

mc = MyBuddy('/dev/ttyACM0', 115200)
time.sleep(1)

REST_LEFT    = [-20.21, 96.94, -17.13, -44.29, 116.36, 23.02]
REST_RIGHT   = [19.16, -82.96, -2.63, 54.75, -121.55, 77]

VIA_LEFT     = [-26.8, 47.81, 78.57, -65.12, 113.81, 28.65]
VIA_RIGHT    = [29.8, -43.15, -81.12, 66.53, -102.83, 98.7]

WAVE_LEFT    = [-24, 49, 75, 57, 159, 15]
WAVE_A_RIGHT = [-159, -45.96, -81.12, -30, 66.18, 106.17]
WAVE_B_RIGHT = [-159, -45.96, -81.12,  70, 66.18, 106.17]

SPEED = 40
WAVE_REPS = 3

def send_both(left, right, speed=SPEED, settle=3.0):
    mc.send_angles(1, left, speed)
    time.sleep(0.1)
    mc.send_angles(2, right, speed)
    time.sleep(settle)

def go_to_rest():
    print("  via-point...")
    send_both(VIA_LEFT, VIA_RIGHT, speed=SPEED, settle=2.5)
    print("  rest...")
    send_both(REST_LEFT, REST_RIGHT, speed=SPEED, settle=4.0)
    print("At rest.")

def wave():
    print("Waving...")
    send_both(WAVE_LEFT, WAVE_A_RIGHT, speed=SPEED, settle=3.0)
    for _ in range(WAVE_REPS):
        mc.send_angles(2, WAVE_A_RIGHT, SPEED)
        time.sleep(1.2)
        mc.send_angles(2, WAVE_B_RIGHT, SPEED)
        time.sleep(1.2)
    print("  returning via via-point...")
    go_to_rest()
    print("Wave complete.")

try:
    mc.power_on()
    time.sleep(2)
    go_to_rest()
    print("Robot ready. Commands: 'hey' | 'quit'")

    while True:
        user_input = input(">> ").strip().lower()
        if user_input == "hey":
            wave()
        elif user_input == "quit":
            print("Shutting down...")
            go_to_rest()
            break
        else:
            print(f"Unknown command: '{user_input}'. Try 'hey' or 'quit'.")

except KeyboardInterrupt:
    print("\nInterrupted — returning to rest")
    go_to_rest()
finally:
    print("Done")
