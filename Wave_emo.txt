# -- Video player state ------------------------------------------
_video_path  = None
_video_lock  = threading.Lock()
_stop_flag   = threading.Event()
_player_thread = None

def _player_loop():
    """Background thread: loops current video until stopped or path changes."""
    cap = None
    current_path = None

    while not _stop_flag.is_set():
        with _video_lock:
            path = _video_path

        if path is None:
            if cap:
                cap.release()
                cap = None
                current_path = None
            time.sleep(0.05)
            continue

        # Switch video if path changed
        if path != current_path:
            if cap:
                cap.release()
            cap = cv2.VideoCapture(path)
            current_path = path
            cv2.namedWindow("MyBuddy", cv2.WINDOW_NORMAL)
            cv2.setWindowProperty("MyBuddy", cv2.WND_PROP_FULLSCREEN, cv2.WINDOW_FULLSCREEN)

        ret, frame = cap.read()
        if not ret:
            cap.set(cv2.CAP_PROP_POS_FRAMES, 0)  # loop
            continue

        cv2.imshow("MyBuddy", frame)
        cv2.waitKey(30)

    # Cleanup on stop
    if cap:
        cap.release()
    cv2.destroyAllWindows()

def start_player():
    global _player_thread
    _stop_flag.clear()
    _player_thread = threading.Thread(target=_player_loop, daemon=True)
    _player_thread.start()


def play_emo(path):
    """Switch to this video immediately ï¿½ loops until changed or stopped."""
    global _video_path
    with _video_lock:
        _video_path = path
    time.sleep(0.1)   # let thread pick up new path


def stop_emo():
    """Stop video and close window."""
    global _video_path
    with _video_lock:
        _video_path = None
    _stop_flag.set()
    if _player_thread:
        _player_thread.join(timeout=2)
    cv2.destroyAllWindows()

# -- Robot helpers -----------------------------------------------

def send_both(left, right, speed=SPEED, settle=3.0):
    mc.send_angles(1, left, speed)
