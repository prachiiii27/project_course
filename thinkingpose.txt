from pymycobot.mybuddy import MyBuddy
import time

mc = MyBuddy('/dev/ttyACM0', 115200)
time.sleep(1)

REST_LEFT      = [-20.21, 96.94, -17.13, -44.29, 116.36, 23.02]
REST_RIGHT     = [19.16, -82.96, -2.63, 54.75, -121.55, 77]

VIA_LEFT       = [-26.8, 47.81, 78.57, -65.12, 113.81, 28.65]
VIA_RIGHT      = [29.8, -43.15, -81.12, 66.53, -102.83, 98.7]

THINKING_LEFT  = [-24.34, 49.65, 75.93, 54.22, 146.6, 22.76]
THINKING_RIGHT = [-28.82, 38.32, 87.09, 60.29, -134.29, 96.6]

SPEED = 40

def send_both(left, right, speed=SPEED, settle=3.0):
    mc.send_angles(1, left, speed)
    time.sleep(0.1)
    mc.send_angles(2, right, speed)
    time.sleep(settle)

def go_to_rest():
    print("  via-point...")
    send_both(VIA_LEFT, VIA_RIGHT, speed=SPEED, settle=2.5)
    print("  rest...")
    send_both(REST_LEFT, REST_RIGHT, speed=SPEED, settle=4.0)
    print("At rest.")

def thinking_posture():
    print("Moving to thinking posture...")
    # rest -> via-point -> thinking
    print("  via-point...")
    send_both(VIA_LEFT, VIA_RIGHT, speed=SPEED, settle=2.5)
    print("  thinking...")
    send_both(THINKING_LEFT, THINKING_RIGHT, speed=SPEED, settle=3.0)
    print("Holding. Press Enter when student is done.")
    input()
    # thinking -> via-point -> rest
    go_to_rest()

try:
    mc.power_on()
    time.sleep(2)
    go_to_rest()
    print("Robot ready. Commands: 'think' | 'quit'")

    while True:
        user_input = input(">> ").strip().lower()
        if user_input == "think":
            thinking_posture()
        elif user_input == "quit":
            print("Shutting down...")
            go_to_rest()
            break
        else:
            print(f"Unknown command: '{user_input}'. Try 'think' or 'quit'.")

except KeyboardInterrupt:
    print("\nInterrupted — returning to rest")
    go_to_rest()
finally:
    print("Done")
